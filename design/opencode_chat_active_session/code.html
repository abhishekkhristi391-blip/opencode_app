export class SSEStreamParser {
  private rawBuffer: string = "";
  private retryCount: number = 0;

  private computeBackoffMs(): number {
    // Full jitter bounded between 100ms and 15s
    const base = Math.min(15000, 100 * Math.pow(2, this.retryCount));
    return Math.floor(Math.random() * base);
  }

  public handleAbort(flushResidual = true): void {
    if (flushResidual && this.rawBuffer.length > 0) {
      this.dispatchPartialChunk(this.rawBuffer);
    }
    this.rawBuffer = "";
    this.retryCount = 0;
  }
}
